name: Android APK Derle ve Yayınla

on:
  push:
    branches: [ main, master ]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - uses: android-actions/setup-android@v3

      - uses: gradle/actions/setup-gradle@v3
        with:
          gradle-version: '8.4'

      - name: Sürüm bilgisi
        run: |
          echo "APP_VERSION_CODE=${{ github.run_number }}" >> $GITHUB_ENV
          echo "APP_VERSION_NAME=1.0.${{ github.run_number }}" >> $GITHUB_ENV

      - name: Derle (debug anahtarıyla imzalı, kurulabilir release)
        run: |
          mkdir -p ~/.android
          [ -f ~/.android/debug.keystore ] || keytool -genkey -v -keystore ~/.android/debug.keystore \
            -storepass android -alias androiddebugkey -keypass android -keyalg RSA -keysize 2048 \
            -validity 10000 -dname "CN=Android Debug,O=Android,C=US"
          gradle :app:assembleRelease --no-daemon --stacktrace
          cp app/build/outputs/apk/release/app-release.apk rgalaxym10-v1.0.${{ github.run_number }}.apk

      - name: GitHub Release oluştur ve APK yükle
        uses: softprops/action-gh-release@v2
        with:
          tag_name: v1.0.${{ github.run_number }}
          name: Tek Tıkla Optimizasyon v1.0.${{ github.run_number }}
          files: rgalaxym10-v1.0.${{ github.run_number }}.apk
          generate_release_notes: true
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
